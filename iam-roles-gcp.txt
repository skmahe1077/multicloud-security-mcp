# GCP IAM Role Assignments for Multi-Cloud Security MCP Server
# Replace SERVICE_ACCOUNT with your service account email
# Replace PROJECT_ID / ORG_ID with your actual values

# Security Command Center Findings Viewer
gcloud organizations add-iam-policy-binding ORG_ID \
  --member="serviceAccount:SERVICE_ACCOUNT" \
  --role="roles/securitycenter.findingsViewer"

# Chronicle SIEM Viewer
gcloud projects add-iam-policy-binding PROJECT_ID \
  --member="serviceAccount:SERVICE_ACCOUNT" \
  --role="roles/chroniclesm.viewer"

# IAM Recommender Viewer
gcloud projects add-iam-policy-binding PROJECT_ID \
  --member="serviceAccount:SERVICE_ACCOUNT" \
  --role="roles/recommender.viewer"

# Cloud DLP Reader
gcloud projects add-iam-policy-binding PROJECT_ID \
  --member="serviceAccount:SERVICE_ACCOUNT" \
  --role="roles/dlp.reader"
